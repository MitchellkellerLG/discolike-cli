    def test_on_result_callback_receives_full_dict(self, manager, mock_client, cache):
        """on_result receives the full poll response dict on each iteration."""
        task_id = "t-on-result"
        cache.save_task(task_id, "discogen")
        mock_client.task_status.side_effect = [
            {"task_id": task_id, "status": "in_progress", "progress": 50, "interim_results": [{"domain": "a.com"}]},
            {"task_id": task_id, "status": "completed", "results": [{"domain": "a.com"}, {"domain": "b.com"}]},
        ]
        results_seen = []
        def on_result(result, attempt, elapsed):
            results_seen.append(result)
        with patch("time.sleep"):
            manager.poll(task_id, on_result=on_result)
        assert len(results_seen) == 2
        assert results_seen[0]["status"] == "in_progress"
        assert "interim_results" in results_seen[0]
        assert results_seen[1]["status"] == "completed"

    **discogen_run pipeline** (mirrors validate pattern from Plan 01):
    1. `client = get_client(ctx)`, `cli_ctx = _get_context(ctx)`
    2. `domains = read_domains(input_file, domain)` then `validate_domain_count(domains)`
    3. Web search warning (per D-12): if `web_search and len(domains) > 50`: `cli_ctx.output.warning(f"Web search on {len(domains)} domains -- this will significantly increase cost. Use --yes to skip this warning.")`
    4. Pre-flight cost estimate (per D-07): Rich Table to stderr showing domain count, context_mode, web_search status, estimated cost range. Use `cli_ctx.cost_tracker.estimate("discogen", len(domains))`.
    5. Confirm gate (same as validate): non-TTY without --yes -> `click.UsageError("--yes flag required in non-interactive mode")`. TTY without --yes -> `click.confirm("Proceed?", default=False, abort=True)`.
    6. Build params: `{"prompt": prompt, "domains": domains, "context_mode": context_mode}`. Add `"web_search": True` if flag set.
    7. Submit: `submit_resp = client.discogen_submit(params)`
    8. Save task: `cli_ctx.cache.save_task(submit_resp["task_id"], "discogen", json.dumps(params))`
    9. Poll with interim display (per D-05, GEN-06):
       - If TTY (`cli_ctx.output.is_tty`): create Rich Live table on stderr, use `on_result` callback to diff `interim_results` and append new rows
       - If not TTY: poll without Live, just use `on_status` for progress percentage to stderr
       - `task_mgr = AsyncTaskManager(client, cli_ctx.cache)`
       - `result = task_mgr.poll(task_id, on_result=on_result_callback)`
    10. Render final results: `cli_ctx.output.render(result.get("results", []), title="DiscoGen Results", cost=cli_ctx.cost_tracker.last_call)`

    **Interim display helper function** `_make_interim_handler(console)`:
    ```python
    def _make_interim_handler(console: Console):
        seen_count = 0
        table = Table(title="Interim Results")
        table.add_column("Domain")
        table.add_column("Result", max_width=60)
        live = Live(table, console=console, refresh_per_second=2)

        def handler(result: dict, attempt: int, elapsed: float) -> None:
            nonlocal seen_count
            interim = result.get("interim_results", [])
            for item in interim[seen_count:]:
                table.add_row(
                    item.get("domain", "?"),
                    str(item.get("result", ""))[:60]
                )
            seen_count = len(interim)

        return live, handler
    ```
